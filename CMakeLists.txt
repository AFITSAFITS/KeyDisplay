cmake_minimum_required(VERSION 3.16)
project(KeyDisplay VERSION 1.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(test_display_state test_display_state.cpp)
if(MSVC)
    target_compile_options(test_display_state PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_display_state PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME display_state COMMAND test_display_state)

if(WIN32)
    add_executable(KeyDisplay WIN32 main.cpp)
    target_link_libraries(KeyDisplay PRIVATE gdiplus gdi32 user32 shell32 ole32 uuid)
    if(MSVC)
        target_compile_options(KeyDisplay PRIVATE /W4 /utf-8)
    else()
        target_compile_options(KeyDisplay PRIVATE -Wall -Wextra -Werror)
        target_link_options(KeyDisplay PRIVATE -static)
    endif()
endif()
